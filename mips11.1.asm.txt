.data
    prompt:      .asciiz "\nDigite um numero (0-100) ou negativo para sair: "
    erro_msg:    .asciiz "Valor invalido! Digite apenas numeros entre 0 e 100."
    msg_0_25:    .asciiz "\n[0-25]: "
    msg_26_50:   .asciiz "\n[26-50]: "
    msg_51_75:   .asciiz "\n[51-75]: "
    msg_76_100:  .asciiz "\n[76-100]: "

.text
.globl main

main:
    # Inicializa contadores ($s0-$s3)
    li $s0, 0
    li $s1, 0
    li $s2, 0
    li $s3, 0

loop:
    # 1. Solicitar e ler o número
    li $v0, 4
    la $a0, prompt
    syscall
    
    li $v0, 5
    syscall
    move $t0, $v0       # $t0 = número digitado

    # 2. Verificar saída (negativo)
    bltz $t0, fim_loop

    # 3. VALIDAÇÃO: Se número > 100, é inválido
    li $t1, 100
    bgt $t0, $t1, mostrar_erro

    # 4. Processar intervalos (Lógica em cascata)
    li $t1, 25
    ble $t0, $t1, inc_0_25
    
    li $t1, 50
    ble $t0, $t1, inc_26_50
    
    li $t1, 75
    ble $t0, $t1, inc_51_75
    
    # Se chegou aqui, sabemos que é entre 76 e 100
    addi $s3, $s3, 1
    j loop

inc_0_25:
    addi $s0, $s0, 1
    j loop

inc_26_50:
    addi $s1, $s1, 1
    j loop

inc_51_75:
    addi $s2, $s2, 1
    j loop

mostrar_erro:
    li $v0, 4
    la $a0, erro_msg
    syscall
    j loop              # Volta para o início sem contar nada

fim_loop:
    # --- Exibição dos Resultados Finais ---
    li $v0, 4
    la $a0, msg_0_25
    syscall
    li $v0, 1
    move $a0, $s0
    syscall

    li $v0, 4
    la $a0, msg_26_50
    syscall
    li $v0, 1
    move $a0, $s1
    syscall

    li $v0, 4
    la $a0, msg_51_75
    syscall
    li $v0, 1
    move $a0, $s2
    syscall

    li $v0, 4
    la $a0, msg_76_100
    syscall
    li $v0, 1
    move $a0, $s3
    syscall

    # Finalizar
    li $v0, 10
    syscall
