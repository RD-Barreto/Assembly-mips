.data
    n1: .asciiz "Digite o primeiro valor: "
    n2: .asciiz "Digite o segundo valor: "
    msg: .asciiz "O resto da divisao do maior pelo menor e: "

.text
.globl main

main:
   
    li $v0, 4
    la $a0, prompt1
    syscall
    li $v0, 5
    syscall
    move $t0, $v0       

    li $v0, 4
    la $a0, prompt2
    syscall
    li $v0, 5
    syscall
    move $t1, $v0       

   
    bge $t0, $t1, calcular
    
    
    move $t2, $t0       # temp = A
    move $t0, $t1       # A = B
    move $t1, $t2       # B = temp

calcular:
   
    div $t0, $t1        
    mflo $t3            
    
   
    mul $t4, $t1, $t3   # $t4 = $t1 * $t3
    
    
    sub $t5, $t0, $t4   # $t5 = $t0 - $t4 (Este é o nosso resto!)

exibir:
    
    li $v0, 4
    la $a0, res_msg
    syscall
    
    move $a0, $t5
    li $v0, 1
    syscall

    
    li $v0, 10
    syscall
