.data
    v:  .asciiz "Digite um numero: "
    r: .asciiz " resultado e: "

.text
.globl main

main:

    li $v0, 4           
    la $a0, v
    syscall
    
    li $v0, 5           
    syscall
    move $t0, $v0     

    li $t1, 2
    rem $t2, $t0, $t1   
    

    beq $t2, $zero, par 

impar:
    addi $t3, $t0, 8    
    j mostrar

par:
    addi $t3, $t0, 5    

mostrar:

    li $v0, 4
    la $a0, r
    syscall
    
    move $a0, $t3       
    li $v0, 1           
    syscall

    li $v0, 10
    syscall
