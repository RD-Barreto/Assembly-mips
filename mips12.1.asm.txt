.data
    mens:      .asciiz "\nDigite a altura da pessoa (em cm): "
    maior:   .asciiz "\nMaior altura: "
    menor:   .asciiz "\nMenor altura: "

.text
.globl main

main:
    
    li $s0, 15          # Contador de pessoas (15 para 1)
    li $s1, 0           # $s1 armazenará a MAIOR altura (começa em 0)
    li $s2, 999         # $s2 armazenará a MENOR altura (começa com valor alto)

loop:
    
    li $v0, 4
    la $a0, mens
    syscall
    
    li $v0, 5           
    syscall
    move $t0, $v0       

    
    ble $t0, $s1, verifica_menor
    move $s1, $t0       
verifica_menor:
    
    bge $t0, $s2, proximo
    move $s2, $t0       
proximo:
    
    addi $s0, $s0, -1   
    bnez $s0, loop      

exibir_resultados:
    
    li $v0, 4
    la $a0, maior
    syscall
    move $a0, $s1
    li $v0, 1
    syscall

    
    li $v0, 4
    la $a0, menor
    syscall
    move $a0, $s2
    li $v0, 1
    syscall

    
    li $v0, 10
    syscall
